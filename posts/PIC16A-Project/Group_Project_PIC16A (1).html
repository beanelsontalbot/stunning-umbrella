  df2 = df[cols].copy()

  return df2

Feature Selection:

Using the exploratory analysis and this function f, we decided on which features to use as predictors in our models.

#function for feature selection
#df must already be processed!

def f(df, model):
  best_score = 0
  best_cols = []
  for col1 in quant_cols:
    for col2 in quant_cols:
      if col1 == col2: continue
      # if score for the model with island, col1, and col2 is > best_score then update best_score and best_cols
      island_score = cross_val_score(model, df[['Island', col1, col2]], df['Species'], cv=5).mean()
      if island_score > best_score:
        best_score = island_score
        best_cols = ['Island', col1, col2]
      # if score for the model with sex, col1, and col2 > best_score then update best_score and best_cols
      sex_score = cross_val_score(model, df[['Sex', col1, col2]], df['Species'], cv=5).mean()
      if sex_score > best_score:
        best_score = cross_val_score(model, df[['Sex', col1, col2]], df['Species'], cv=5).mean()
        best_cols = ['Sex', col1, col2]
  #plot model?
  return best_score, best_cols

Modeling:

We chose complexity parameters and then ran the models and made decision regions and confusion matrices for both training and testing data to evaluate the models.

#function to plot decision regions
#X should only include selected features and be processed
from matplotlib.colors import ListedColormap

def plot_regions(model, X, y, features, ax):
  """
  a function to plot decision regions for the model given.
  Arguments: model is the model, X and y are the predictor and target data sets, respectively. features is a list of the features used to predict the target data. ax is the axis
  to plot the decision regions on. Note: features should have the QUALITATIVE column at the beginning, and the data should already be grouped by the qualitative column, and
  the model trained based on the grouped data.
  Returns: None
  """
# for convenience, give names to the columns of the data
  x0 = X[features[1]]
  x1 = X[features[2]]
# create a grid
  grid_x = np.linspace(x0.min() - 1,x0.max() + 1,501)
  grid_y = np.linspace(x1.min() - 1,x1.max() + 1,501)
  xx, yy = np.meshgrid(grid_x, grid_y)
# extract model predictions
  XX = xx.ravel()
  YY = yy.ravel()
  XY = pd.DataFrame({features[0]: X[features[0]].iloc[0], f"{features[1]}" : XX, f"{features[2]}" : YY})
  p = model.predict(XY)
  p = p.reshape(xx.shape)
# create the colormap to match species to color consistently
  colors = [species_colors_num[species] for species in model.classes_]  # Assuming model.classes_ contains species names
  cmap = ListedColormap(colors)
# Use the new colormap for contour and scatter plots
  ax.contourf(xx, yy, p, cmap=cmap, alpha=0.2, vmin=0, vmax=len(species_colors_num) - 1)
  ax.scatter(x0, x1, c=y, cmap=cmap, vmin=0, vmax=len(species_colors_num) - 1)
  for species, color in species_colors.items():
    ax.scatter([], [], color=color, label=species)  # Creating dummy points for legend
  ax.set(xlabel = features[1], ylabel = features[2], title = f"{X[features[0]].iloc[0]}")
  ax.legend(title = 'Species')

Discussion:

The random forest model is a set of decision trees on subsets of the data that get averaged to make an overall model that is better for overfitting. I chose my model based on recommendations from the explorations I did in the exploratory analysis and the functions I wrote to test different options. I used cross validation to determine the complexity parameter (best max_depth) and the function f for feature selection. (Sometimes f will give that the best qualitative column is Island, but most of the time it was Sex, and I could see from the table in the exploratory analysis section that Sex was strongly correlated to these variables, so that is what I chose for my model). As we can see, the random forest model did a pretty good job on both the training and testing data sets (of course, this makes sense, since the cross validation score was very high for both data sets when we tested it in the complexity parameter stage). Since it doesn’t look like there is too much overfitting on the decision regions, and since it did well with the testing data, I think this model would do well if more data was added. There are a few points that the model almost got wrong, so it would probably not be perfect for more data points.
